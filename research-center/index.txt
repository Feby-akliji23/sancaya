  var currentPageId = null;

  function getPageId() {
    var root = document.querySelector('[data-acc-root]');
    if (!root) return null;
    if (!root.dataset.accId) {
      root.dataset.accId = 'page-' + Date.now();
    }
    return root.dataset.accId;
  }

  function openAccordion() {
    var hash = location.hash.slice(1);
    if (!hash) return;

    var pageId = getPageId();
    if (!pageId) return;

    if (currentPageId !== pageId) {
      currentPageId = pageId;
    }

    var targetId = decodeURIComponent(hash);
    var attempts = 0;

    function tryOpen() {
      if (++attempts > 30) return;
      if (getPageId() !== pageId) return;

      var element = document.getElementById(targetId);
      if (!element) { setTimeout(tryOpen, 100); return; }

      var details = element.closest('details');
      if (!details) return;

      var root = details.closest('[data-acc-root]');
      if (!root || root.dataset.accId !== pageId) return;

      root.querySelectorAll('details[open]').forEach(function(d){ if(d!==details) d.open=false; });
      details.open = true;

      var summary = details.querySelector('summary');
      if (summary && summary.focus) { try { summary.focus({ preventScroll: true }); } catch(_) {} }

      setTimeout(function() {
        if (getPageId() === pageId) {
          var rect = details.getBoundingClientRect();
          var scrollTop = window.pageYOffset || document.documentElement.scrollTop;
          var targetPos = rect.top + scrollTop - 120; // header offset
          window.scrollTo({ top: Math.max(0, targetPos), behavior: 'smooth' });
        }
      }, 150);
    }

    tryOpen();
  }

  function init() {
    currentPageId = getPageId();
    setTimeout(openAccordion, 150);
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init);
  else init();

  window.addEventListener('hashchange', openAccordion);
  window.addEventListener('pageshow', function(){ setTimeout(openAccordion, 150); });
  window.addEventListener('popstate', function(){ setTimeout(openAccordion, 150); });

  new MutationObserver(function() {
    var root = document.querySelector('[data-acc-root]');
    if (root) {
      var newId = getPageId();
      if (newId !== currentPageId) { init(); }
    }
  }).observe(document.body, { childList: true, subtree: true });
